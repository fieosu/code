# 多网址监控（Multi-URL Monitoring）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把当前单 URL 监控扩展为支持同时监控多个 Downdetector 状态页：每轮共享一个浏览器串行捕获所有站，逐站检测、逐站独立推送告警（消息带站名 + 该站截图）。

**Architecture:** 配置层 `config.TARGETS` 定义站点列表（`name + url + 可选 reports_threshold`），`TARGETS` 为空时回落 `TARGET_URL` 单站。`screenshot.py` 抽出页面级函数 `_capture_page()` 供单站与多站共用，新增 `capture_all()` 在一个持久化上下文里串行处理所有站。`monitor.py` 遍历目标，每站 `detect()`（带该站基准图路径与该站阈值）→ 异常则 `notify()`（带站名）。

**Tech Stack:** Python 3、patchright（Playwright fork）、Pillow、numpy、scikit-image、requests。无测试框架。

## Global Constraints

- 代码注释与日志使用中文（遵循项目约定）。
- Windows 下运行需前缀 `PYTHONIOENCODING=utf-8`。
- 本目录不是 git 仓库：**不做 commit 步骤**。
- 无测试框架：纯函数逻辑用 `python -c` 断言验证；浏览器流程用实际运行 `monitor.py --baseline` / `--once` 验证。
- `import patchright`，绝不 `import playwright`。
- 不删除 `.browser_profile/`（CF cookie 缓存）。
- 单站兼容：`TARGETS` 为空时必须继续走 `TARGET_URL` 单站模式，现有行为不回归。
- 每站失败必须隔离：一个站出错只记日志，不影响其余站与本轮整体。

```python
def detect(screenshot_path: str, page_text: str,
           baseline_path: str = None, reports_threshold: int = None) -> tuple[bool, str]:
    """
    综合检测入口。

    Args:
        screenshot_path: 当前截图路径
        page_text: 页面可见文本
        baseline_path: 该站基准图路径；None 用 config.BASELINE_IMAGE
        reports_threshold: 该站报告数阈值；None 用 config.REPORTS_THRESHOLD

    Returns:
        (是否异常, 异常原因)
    """
    results = []

    # Layer 1
    abnormal, reason = check_by_keywords(page_text)
    results.append(("Layer 1 关键词", abnormal, reason))

    # Layer 2
    abnormal, reason = check_by_image(screenshot_path, baseline_path)
    results.append(("Layer 2 图像SSIM", abnormal, reason))

    # Layer 3
    abnormal, reason = check_by_reports(page_text, reports_threshold)
    results.append(("Layer 3 报告数", abnormal, reason))

    # 汇总各层结果（保留 Layer 1→3 的优先级，取最先命中的原因）
    logger.info("检测结果:")
    hit_reason = None
    for name, is_abnormal, reason in results:
        if is_abnormal:
            hit_reason = reason
            logger.info(f"  {name}: ❌ {reason}")
        else:
            logger.info(f"  {name}: ✅ 正常")

    if hit_reason:
        logger.warning(f"🚨 异常检测到: {hit_reason}")
        return True, hit_reason

    logger.info("✅ 页面正常，未检出异常")
    return False, ""
```

```python
def _capture_page(page, target, output_path, *, headless, want_text):
    """
    在已打开的 page 上完成单个目标截图全流程：访问 → CF → 等渲染 → 悬停提取
    tooltip → 隐藏 fixed 浮层 → 截图 → 裁剪 → 取文本。

    由单站入口 _capture_once 和多站入口 capture_all 共用：多站时同一浏览器
    上下文/页面串行调用本函数。

    Returns:
        (output_path, text|None)；headless 下被 CF 拦截需回退有头时返回 None。
    """
    cf_headless_timeout = getattr(config, "CF_HEADLESS_TIMEOUT", 15)
    cf_headed_timeout = getattr(config, "CF_HEADED_TIMEOUT", 120)
    network_idle_timeout = getattr(config, "NETWORKIDLE_TIMEOUT", 10)

    try:
        logger.info(f"正在访问 {target} (headless={headless}) ...")
        page.goto(target, wait_until="domcontentloaded", timeout=60000)

        if _is_cloudflare_challenge(page):
            logger.warning("检测到 Cloudflare 验证")
            if headless:
                # headless 下只能等非交互式 managed challenge 自动放行；
                # 一旦是交互式挑战，headless 必然过不了，直接回退有头。
                passed = _wait_for_clearance(page, cf_headless_timeout)
                if not passed:
                    logger.warning("headless 未能通过 Cloudflare，将切到有头模式刷新 cookie")
                    return None
            else:
                # 有头模式：先等自动通过，过半时间还没过就尝试点击兜底
                half = cf_headed_timeout // 2
                logger.info(f"等待 Cloudflare 自动通过（最多 {cf_headed_timeout}s）...")
                passed = _wait_for_clearance(page, half)
                if not passed:
                    logger.warning("未自动通过，尝试点击验证框 ...")
                    _try_click_turnstile(page)
                    passed = _wait_for_clearance(page, cf_headed_timeout - half)
                if not passed:
                    logger.error(
                        "有头模式仍未通过 Cloudflare（可能网络异常或该站为交互式验证）"
                    )

        # 等待页面渲染完成
        try:
            page.wait_for_load_state("networkidle", timeout=network_idle_timeout * 1000)
        except PlaywrightTimeout:
            logger.debug("networkidle 超时，按当前状态继续")
        page.wait_for_timeout(2000)
        try:
            logger.info(f"页面标题: {page.title()}")
        except Exception:
            pass

        # 悬停图表让 tooltip（含报告数）渲染——先悬停再截图，数字才会出现在图片里
        chart_bottom = _hover_chart_for_tooltip(page)

        # 隐藏 fixed 浮层，避免吸顶导航被全页截图画到文档中部（见 _hide_fixed_ui）
        _hide_fixed_ui(page)

        page.screenshot(path=output_path, full_page=config.SCREENSHOT_FULL_PAGE)
        _crop_screenshot(output_path, chart_bottom)
        text = page.inner_text("body") if want_text else None
        logger.info(f"截图已保存 -> {output_path}")
        return output_path, text

    except PlaywrightTimeout:
        logger.warning("页面加载超时，尝试当前状态截图")
        chart_bottom = _hover_chart_for_tooltip(page)
        _hide_fixed_ui(page)
        try:
            page.screenshot(path=output_path, full_page=config.SCREENSHOT_FULL_PAGE)
            _crop_screenshot(output_path, chart_bottom)
        except Exception:
            pass
        text = None
        if want_text:
            try:
                text = page.inner_text("body")
            except Exception:
                text = None
        return output_path, text
```

```python
def capture_all(targets: list) -> list:
    """
    多站：启动一个持久化浏览器上下文，串行捕获所有目标。

    所有站同域（downdetector.com），共享同一个 cf_clearance：CF 只过一次，
    首个站放行后其余站直连。每站失败只记日志并继续，整轮不中断。
    固定走屏幕外有头窗口（生产路径，与默认单站策略一致）。

    Args:
        targets: 形如 [{"name": "amazon", "url": "..."}, ...] 的列表

    Returns:
        与输入等长的 [(name, output_path|None, page_text|"")]，失败项 output_path 为 None
    """
    if not targets:
        return []

    results = []
    with sync_playwright() as p:
        launch_args = ["--no-sandbox"]
        if getattr(config, "CF_HEADED_OFFSCREEN", True):
            launch_args.append("--window-position=-32000,-32000")
        context = p.chromium.launch_persistent_context(
            user_data_dir=BROWSER_PROFILE_DIR,
            headless=False,
            viewport={
                "width": config.VIEWPORT_WIDTH,
                "height": config.VIEWPORT_HEIGHT,
            },
            args=launch_args,
        )
        page = context.pages[0] if context.pages else context.new_page()
        try:
            for t in targets:
                name = t.get("name") or "site"
                output_path = f"capture_{name}.png"
                try:
                    result = _capture_page(page, t["url"], output_path, headless=False, want_text=True)
                    results.append((name, result[0], result[1] or ""))
                except Exception as e:
                    logger.error(f"[{name}] 捕获失败: {e}")
                    results.append((name, None, ""))
        finally:
            context.close()
    return results
```

```python
def _build_message(site_name: str, reason: str) -> str:
    """拼接告警文本：基础消息 + 站点 + 异常原因。"""
    msg = config.NOTIFY_MESSAGE
    lines = []
    if site_name:
        lines.append(f"站点: {site_name}")
    if reason:
        lines.append(f"异常原因: {reason}")
    if lines:
        msg += "\n\n" + "\n".join(lines)
    return msg


def notify(image_path: str, reason: str = "", site_name: str = ""):
    """
    推送告警到配置的 Webhook。

    Args:
        image_path: 异常截图路径
        reason: 异常原因（附在消息里）
        site_name: 站点名（附在消息里）

    sender = _SENDERS.get(sender_name)
    if not sender:
        logger.error(f"不支持的 Webhook 类型: {config.WEBHOOK_TYPE}")
        return

    msg = _build_message(site_name, reason)

    logger.info(f"正在推送告警 → {sender_name} ...")
    sender(image_path, msg)
    logger.info("告警推送成功")
```

```python
def run_once(cycle: int = None) -> bool:
    """
    执行单次检测循环：捕获所有站 → 逐站三层检测 → 异常站逐站推送告警。

    Args:
        cycle: 第几轮巡检（持续模式递增；单次运行可不传）

    Returns:
        是否有任一站点异常
    """
    label = f"巡检 #{cycle}" if cycle else "巡检"
    start = datetime.now()
    logger.info("=" * 50)
    logger.info(f"{label} 开始 — {start.strftime('%Y-%m-%d %H:%M:%S')}")

    targets = get_targets()
    logger.info(f"本轮共 {len(targets)} 个站点")

    # 1) 一个浏览器串行捕获所有站
    captures = capture_all(targets)          # [(name, path|None, text|"")]

    # 2) 逐站检测 + 异常推送
    abnormal_count = 0
    for t, (name, path, text) in zip(targets, captures):
        logger.info(f"── 站点: {name} ({t['url']}) ──")
        if not path:
            logger.error(f"[{name}] 截图失败，跳过检测")
            continue
        baseline = f"baseline/{name}.png"
        threshold = t.get("reports_threshold")     # None → detect 内回落全局
        abnormal, reason = detect(path, text, baseline_path=baseline, reports_threshold=threshold)
        if abnormal:
            abnormal_count += 1
            notify(path, reason, site_name=name)

    elapsed = (datetime.now() - start).total_seconds()
    logger.info(
        f"{label} 结束 — 异常站点 {abnormal_count}/{len(targets)}，耗时 {elapsed:.1f}s"
    )
    logger.info("=" * 50)
    return abnormal_count > 0
```
